## Troubleshoot
kubectl describe pod <pod> -n <ns>
kubectl describe deployment <deploy> -n <ns>
kubectl logs <pod> -n <ns> --tail 100
kubectl logs <pod> -n <ns> -c <container> --previous
kubectl get events -n <ns> --sort-by=.lastTimestamp
kubectl top pods -n <ns> --sort-by=memory
kubectl top nodes

## Rollout scaling
kubectl rollout status  deployment/<name> -n <ns>
kubectl rollout history deployment/<name> -n <ns>
kubectl rollout restart deployment/<name> -n <ns>
kubectl rollout undo    deployment/<name> -n <ns>
kubectl scale   deployment/<name> --replicas=3 -n <ns>
kubectl delete  pod <pod> -n <ns>
kubectl cordon   <node>
kubectl uncordon <node>

## Context (cluster switch)
kubectl config current-context              # show active cluster in this channel
kubectl config get-contexts                 # list clusters the bot can target
kubectl config use-context <cluster>        # switch active cluster for this channel
kubectl <cmd> --cluster <name>              # one-off override

## Filter output(post-processing)
kubectl get pods -A | grep <pattern>
kubectl get pods -A | grep -v Running
kubectl get pods -A | grep -i error

## kind
kind create cluster --name shoplite
kind get clusters               # shows real active cluster
kind-<clustername>
kubectl config get-contexts     # shows ALL remembered clusters (even stale ones)
kubectl config use-context kind-shoplite    # selects the one you actually use
kubectl config delete-context <clustername>
kubectl cluster-info --context kind-
